HTML <head> Snippet.

Add this to 02-woodmart-child/functions.php via the wp_head hook. This covers every device, every OS, every browser.

<!-- Performance Peptides — full favicon & icon pack -->
<link rel="shortcut icon" href="/wp-content/themes/woodmart-child/assets/icons/favicon.ico">
<link rel="icon" type="image/png" sizes="16x16" href="/wp-content/themes/woodmart-child/assets/icons/icon-16.png">
<link rel="icon" type="image/png" sizes="32x32" href="/wp-content/themes/woodmart-child/assets/icons/icon-32.png">
<link rel="icon" type="image/png" sizes="192x192" href="/wp-content/themes/woodmart-child/assets/icons/icon-192.png">
<link rel="icon" type="image/png" sizes="512x512" href="/wp-content/themes/woodmart-child/assets/icons/icon-512.png">
